cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AeroSimConfiguration LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 14)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_AUTOMOC ON)
    if(MSVC)
        add_compile_options(/utf-8)
    endif()
    find_package(Qt5 5.12.12 EXACT REQUIRED COMPONENTS Core)
    option(AEROSIM_BUILD_TESTS "Build configuration tests" ON)
    if(AEROSIM_BUILD_TESTS)
        enable_testing()
    endif()
    add_subdirectory(../../contracts contracts)
endif()
find_package(Qt5 5.12.12 EXACT REQUIRED COMPONENTS Core Xml)
add_library(aerosim_configuration STATIC src/XcfProjectDocument.cpp src/CalculationControlConfig.cpp src/OutputControlConfig.cpp src/SolutionMethodConfig.cpp)
add_library(aerosim::configuration ALIAS aerosim_configuration)
target_include_directories(aerosim_configuration PUBLIC include)
target_link_libraries(aerosim_configuration PUBLIC aerosim_contracts Qt5::Core Qt5::Xml)
if(MSVC)
    target_compile_options(aerosim_configuration PRIVATE /utf-8)
endif()
if(AEROSIM_BUILD_TESTS)
    add_executable(aerosim_xcf_test tests/XcfProjectDocumentTest.cpp)
    target_link_libraries(aerosim_xcf_test PRIVATE aerosim_configuration)
    add_test(NAME xcf_project_roundtrip COMMAND aerosim_xcf_test)
    add_executable(aerosim_solution_method_test tests/SolutionMethodTest.cpp)
    target_link_libraries(aerosim_solution_method_test PRIVATE aerosim_configuration)
    add_test(NAME xcf_solution_method COMMAND aerosim_solution_method_test)
endif()
